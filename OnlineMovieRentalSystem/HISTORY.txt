CREATE TABLE History(
    username VARCHAR(50),
	title VARCHAR(50),
	rentalDate VARCHAR(50),
	returnDate VARCHAR(50),
	price DECIMAL(6, 2),
	payback DECIMAL(6, 2),
	paychange DECIMAL(6, 2),
        extra DECIMAL(6, 2) DEFAULT 0,
	extrapay DECIMAL(6, 2) DEFAULT 0,
	extrachange DECIMAL(6, 2) DEFAULT 0,
	total DECIMAL(6, 2),
    FOREIGN KEY (username) REFERENCES LoginUser(username) ON DELETE CASCADE
);
